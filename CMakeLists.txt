cmake_minimum_required(VERSION 3.20)
project(quantumshield_crypto_bench VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto)
add_executable(qs-crypto-bench src/main.cpp)
target_link_libraries(qs-crypto-bench PRIVATE OpenSSL::Crypto)
target_compile_options(qs-crypto-bench PRIVATE -Wall -Wextra -Wpedantic)
